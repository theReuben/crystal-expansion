# D114: the start menu's POKeGEAR entry and the Pokedex's Johto tilemaps.
#
# The dex refuses to open with nothing seen (HandleStartMenuInput checks
# GetNationalPokedexCount), so this fakes 64 seen/caught species. The two
# offsets into SaveBlock1 are the ones GetSetPokedexFlag carries in its own
# constant pool -- dexSeen 0x32E8, dexCaught 0x3369 -- and will move if the
# expansion's FREE_* save options change; 0x0300523C is gSaveBlock1Ptr, spelled
# out because raw pwrite lines are not symbol-resolved.
include newgame.txt

setflag FLAG_SYS_POKEGEAR_GET
setflag FLAG_SYS_POKEDEX_GET
pwrite 0x0300523c 0x32E8 32 0xFFFFFFFF
pwrite 0x0300523c 0x32EC 32 0xFFFFFFFF
pwrite 0x0300523c 0x3369 32 0xFFFFFFFF
pwrite 0x0300523c 0x336D 32 0xFFFFFFFF

press START
wait 40
shot menu

press A
wait 180
shot dex
